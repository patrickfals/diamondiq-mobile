import { useState, useMemo } from "react";
import { View, Text, TouchableOpacity, StyleSheet } from "react-native";
import { COLORS } from "../theme/colors";
import rawCache from "../data/commentary_cache.json";

// --- Types -------------------------------------------------------------------

type KnowledgeLevel = "novice" | "learning" | "regular" | "expert";

interface CacheEntry {
  commentary_by_level: Record<KnowledgeLevel, string>;
  evidence_pitches: number[];
  complete: boolean;
  elapsed_seconds: number;
  edited?: boolean;
  edit_note?: string;
}

const cache = rawCache as Record<string, CacheEntry>;

// --- Level metadata ----------------------------------------------------------

const LEVELS: { key: KnowledgeLevel; label: string; description: string }[] = [
  { key: "novice",   label: "Novice",   description: "New to baseball" },
  { key: "learning", label: "Learning", description: "Getting started" },
  { key: "regular",  label: "Regular",  description: "Familiar with the game" },
  { key: "expert",   label: "Expert",   description: "Deep baseball knowledge" },
];

// --- Props -------------------------------------------------------------------

interface CommentaryPanelProps {
  /** replay.state.revealedCount — 0 means nothing revealed yet */
  revealedCount: number;
}

// --- Component ---------------------------------------------------------------

export default function CommentaryPanel({ revealedCount }: CommentaryPanelProps) {
  const [level, setLevel] = useState<KnowledgeLevel>("learning");

  // Only use an entry when complete === true; treat incomplete as unavailable.
  const entry: CacheEntry | null = useMemo(() => {
    if (revealedCount === 0) return null;
    const found = cache[String(revealedCount)];
    return found?.complete === true ? found : null;
  }, [revealedCount]);

  const commentaryText: string | null = entry
    ? entry.commentary_by_level[level] ?? null
    : null;

  // -- No pitches revealed yet: verified intro, no spoilers ------------------
  if (revealedCount === 0) {
    return (
      <View style={styles.container}>
        <Text style={styles.sectionTitle}>STORY OF THE AT-BAT</Text>
        <LevelSelector current={level} onChange={setLevel} />
        <View style={styles.card}>
          <Text style={styles.introText}>
            Trevor Story vs. Michael Soroka. Story comes in hitting .244 on the
            season; Soroka carries a 4.70 ERA. Press Play to watch the at-bat
            unfold pitch by pitch.
          </Text>
        </View>
      </View>
    );
  }

  // -- Entry unavailable for this count -------------------------------------
  if (!entry || !commentaryText) {
    return (
      <View style={styles.container}>
        <Text style={styles.sectionTitle}>STORY OF THE AT-BAT</Text>
        <LevelSelector current={level} onChange={setLevel} />
        <View style={styles.card}>
          <Text style={styles.placeholder}>
            No commentary available for pitch {revealedCount}.
          </Text>
        </View>
      </View>
    );
  }

  // -- Commentary available -------------------------------------------------
  return (
    <View style={styles.container}>
      <Text style={styles.sectionTitle}>STORY OF THE AT-BAT</Text>
      <LevelSelector current={level} onChange={setLevel} />
      <View style={styles.card}>
        <Text style={styles.commentaryText}>{commentaryText}</Text>
      </View>
    </View>
  );
}

// --- Level selector ----------------------------------------------------------

function LevelSelector({
  current,
  onChange,
}: {
  current: KnowledgeLevel;
  onChange: (l: KnowledgeLevel) => void;
}) {
  const activeEntry = LEVELS.find((l) => l.key === current);

  return (
    <View style={styles.levelBlock}>
      <View style={styles.levelRow}>
        {LEVELS.map(({ key, label }) => {
          const active = key === current;
          return (
            <TouchableOpacity
              key={key}
              onPress={() => onChange(key)}
              style={[styles.levelTab, active && styles.levelTabActive]}
              accessibilityRole="button"
              accessibilityState={{ selected: active }}
              accessibilityLabel={`${label} commentary level`}
            >
              <Text style={[styles.levelTabText, active && styles.levelTabTextActive]}>
                {label}
              </Text>
            </TouchableOpacity>
          );
        })}
      </View>
      {activeEntry && (
        <Text style={styles.levelDescription}>{activeEntry.description}</Text>
      )}
    </View>
  );
}

### 5. The signed `run_id` approach — assessment and backward compatibility

**Proposal:** extend the token payload from `game_id:at_bat_index:revealed_count` to `game_id:at_bat_index:revealed_count:run_id`.

**Backward compatibility analysis:**

The current `verify_snapshot_token` splits the decoded raw string on `"."` into exactly 2 parts (payload + signature) and then splits the payload on `":"` expecting exactly 3 fields. Adding a 4th `:` to the payload is non-breaking at the outer `.` level but **will break the inner** `if len(fields) != 3` check in `verify_snapshot_token`.

Fix: change the inner check to `if len(fields) not in (3, 4)` and treat `fields[3]` as optional `run_id`. Existing 3-field tokens continue to verify; new 4-field tokens also verify. The `run_id` is extracted from the payload if present, otherwise `None`.

**IBM OpenAPI contract concern:** the snapshot token is an opaque string passed by IBM Pitch Scout as a parameter to `POST /snapshot/redeem`. The OpenAPI spec defines the `token` field as a string with no length or character constraints. The deployed Render endpoint still accepts any valid base64url string; the internal structure is invisible to IBM. **No OpenAPI contract change is needed.**

**Security note:** `run_id` is a short random hex string (8–16 chars), not a secret. It provides correlation, not authentication. If someone constructed a token with a matching `run_id` they already know, they still cannot pass the HMAC check without the `SNAPSHOT_SECRET`. The `run_id` changes what is signed; a token with a forged `run_id` would have an invalid signature.

**Conclusion:** this approach is sound and low-risk. Both the local signer (precompute script) and the deployed Render verifier must be updated before the pilot.

### 6. Correlation method and limitations

The correlation chain:

1. **Precompute script** generates `run_id = secrets.token_hex(8)` for each position execution.
2. **Token creation**: `create_snapshot_token(game_id, at_bat_index, revealed_count, run_id=run_id)` embeds it in the HMAC-signed payload.
3. **IBM Analyst** receives the token in the prompt, passes it to Pitch Scout.
4. **Pitch Scout** calls `POST /snapshot/redeem` with the token.
5. **Render** verifies the HMAC, extracts `run_id` from the payload, records it in the in-memory history.
6. **Precompute script** queries `GET /snapshot/redemptions/recent` after the IBM call completes and finds the matching `run_id`.

**What this proves:** the signed token created by the precompute script was redeemed by some caller at the recorded timestamp. The HMAC signature guarantees the token was not tampered with after creation.

**What this does NOT prove:** which specific IBM agent (Analyst vs. Pitch Scout vs. a human) initiated the HTTP call to `/snapshot/redeem`. Without IBM Observability records, internal delegation cannot be verified in the trace.

**Clock caveat:** the precompute script measures `time.monotonic()` for elapsed duration (accurate for local duration). UTC timestamps from Render (redemption event) and local UTC timestamps (token creation) may differ by NTP skew. The trace will record both and note this limitation.

def fetch_game_data(game_id: int, timeout: int = 15) -> dict:
    """

    Parameters

    game_id : MLB game identifier (must be > 0)

def extract_batter_stats(boxscore_teams: dict, batter_id) -> dict | None:
    """
    Compute entering-game batting statistics for the batter.

    Uses the postgame seasonStats subtraction method: entering value =
    season total - this game's contribution.  Returns a stats dict on
    success, or None if any value is missing, the denominator is zero,
    or the intermediate totals are impossible.
    """
    batter_player = None
    for side in ("home", "away"):
        players = boxscore_teams.get(side, {}).get("players", {})
        key = f"ID{batter_id}"
        if key in players:
            batter_player = players[key]
            break

    if batter_player is None:
        return None

    season_bat = batter_player.get("seasonStats", {}).get("batting", {})
    game_bat   = batter_player.get("stats", {}).get("batting", {})

    season_hits = season_bat.get("hits")
    season_ab   = season_bat.get("atBats")
    game_hits   = game_bat.get("hits")
    game_ab     = game_bat.get("atBats")

    if any(v is None for v in (season_hits, season_ab, game_hits, game_ab)):
        return None

    entering_hits = int(season_hits) - int(game_hits)
    entering_ab   = int(season_ab)   - int(game_ab)

    if entering_hits < 0 or entering_ab < 0:
        return None
    if entering_ab == 0:
        return None
    if entering_hits > entering_ab:
        return None

    avg = round(entering_hits / entering_ab, 3)
    return {
        "avg":     avg,
        "hits":    entering_hits,
        "at_bats": entering_ab,
        "basis":   "entering_game",
    }


def extract_pitcher_stats(boxscore_teams: dict, pitcher_id) -> dict | None:
    """
    Compute entering-game pitching statistics for the pitcher.

    Uses the postgame seasonStats subtraction method.  The MLB API outs field
    is used directly (whole-number out count, not innings-pitched decimal).

    Returns a stats dict on success, or None if any value is missing,
    the denominator is zero, or the totals are impossible.
    """
    pitcher_player = None
    for side in ("home", "away"):
        players = boxscore_teams.get(side, {}).get("players", {})
        key = f"ID{pitcher_id}"
        if key in players:
            pitcher_player = players[key]
            break

    if pitcher_player is None:
        return None

    season_pit = pitcher_player.get("seasonStats", {}).get("pitching", {})
    game_pit   = pitcher_player.get("stats", {}).get("pitching", {})

    season_er   = season_pit.get("earnedRuns")
    season_outs = season_pit.get("outs")
    game_er     = game_pit.get("earnedRuns")
    game_outs   = game_pit.get("outs")

    if any(v is None for v in (season_er, season_outs, game_er, game_outs)):
        return None

    entering_er   = int(season_er)   - int(game_er)
    entering_outs = int(season_outs) - int(game_outs)

    if entering_er < 0 or entering_outs < 0:
        return None
    if entering_outs == 0:
        return None

    era = round(27 * entering_er / entering_outs, 2)
    return {
        "era":           era,
        "earned_runs":   entering_er,
        "outs_recorded": entering_outs,
        "basis":         "entering_game",
    }

            f"pos {revealed_count}: expected {revealed_count} pitches, "
            f"got {len(result['pitches'])}")

        self.assertNotIn("total_pitches", result,
            f"pos {revealed_count}: total_pitches must not appear in partial response")

        # is_complete must be False
        self.assertFalse(result["is_complete"],
            f"pos {revealed_count}: is_complete must be False")

import base64
import hashlib
import hmac
import os
import re as _re

# Only lowercase hex characters are accepted in run_id.
_RUN_ID_RE = _re.compile(r'^[0-9a-f]+$')


def _get_secret() -> bytes:
    """
    Read SNAPSHOT_SECRET from the environment.

    Raises ValueError — not KeyError — so callers can produce a 503 response
    without matching on exception type.  The ValueError message does not
    include the secret value.
    """
    secret = os.environ.get("SNAPSHOT_SECRET")
    if not secret:
        raise ValueError("SNAPSHOT_SECRET environment variable is not set")
    return secret.encode("utf-8")


def _build_payload(
    game_id: int,
    at_bat_index: int,
    revealed_count: int,
    run_id: str | None = None,
) -> str:
    """
    Build the canonical payload string.

    With run_id=None  → "game_id:at_bat_index:revealed_count"           (3-field)
    With run_id=<str> → "game_id:at_bat_index:revealed_count:run_id"    (4-field)

    run_id must be a non-empty lowercase hex string.  Colons are unambiguous
    delimiters because the integer fields contain only digits and run_id
    contains only hex characters (0-9, a-f); none of these include ':' or '.'.
    """
    if run_id is not None:
        if not run_id:
            raise ValueError("run_id must not be empty")
        if not _RUN_ID_RE.match(run_id):
            raise ValueError(
                f"run_id must be a non-empty lowercase hex string, got: {run_id!r}"
            )
        return f"{game_id}:{at_bat_index}:{revealed_count}:{run_id}"
    return f"{game_id}:{at_bat_index}:{revealed_count}"


def _sign_payload(payload: str, secret: bytes) -> str:
    """Return the lowercase hex HMAC-SHA256 digest of a UTF-8 payload."""
    return hmac.new(secret, payload.encode("utf-8"), hashlib.sha256).hexdigest()


def create_snapshot_token(
    game_id: int,
    at_bat_index: int,
    revealed_count: int,
    run_id: str | None = None,
) -> str:
    """
    Create a signed snapshot token for the given baseball identifiers.

    Parameters
    ----------
    game_id        : MLB game identifier (must be > 0)
    at_bat_index   : zero-based at-bat index (must be >= 0)
    revealed_count : number of pitches authorized (must be >= 0)
    run_id         : optional non-secret run correlation ID.
                     Must be a non-empty lowercase hex string when provided
                     (e.g. secrets.token_hex(8) → 16-char hex string).
                     Embedded in the signed payload; not a secret.

    Returns
    -------
    A URL-safe base64-encoded token (no padding) that encodes the payload
    and its HMAC-SHA256 signature.

    Raises
    ------
    ValueError : if any input is out of range, run_id is invalid,
                 or SNAPSHOT_SECRET is unset.
    """
    if game_id <= 0:
        raise ValueError(f"game_id must be > 0, got {game_id}")
    if at_bat_index < 0:
        raise ValueError(f"at_bat_index must be >= 0, got {at_bat_index}")
    if revealed_count < 0:
        raise ValueError(f"revealed_count must be >= 0, got {revealed_count}")

    secret = _get_secret()
    payload = _build_payload(game_id, at_bat_index, revealed_count, run_id)
    signature = _sign_payload(payload, secret)
    raw = f"{payload}.{signature}"
    return base64.urlsafe_b64encode(raw.encode("utf-8")).rstrip(b"=").decode("ascii")


def verify_snapshot_token(
    token: str,
) -> tuple[int, int, int, str | None]:
    """
    Verify a snapshot token and return the encoded identifiers.

    Parameters
    ----------
    token : the opaque string returned by create_snapshot_token

    Returns
    -------
    (game_id, at_bat_index, revealed_count, run_id)

    run_id is None for legacy 3-field tokens.
    run_id is a non-empty lowercase hex string for 4-field tokens.

    Raises
    ------
    ValueError      : token is malformed (wrong structure, non-integer fields,
                      invalid run_id format, or values out of range)
    PermissionError : HMAC signature does not match (tampered token)
    ValueError      : SNAPSHOT_SECRET is not set (message indicates config error)
    """
    # --- Decode base64 (re-add stripped padding) ---
    try:
        padded = token + "=" * (-len(token) % 4)
        raw = base64.urlsafe_b64decode(padded.encode("ascii")).decode("utf-8")
    except Exception:
        raise ValueError("Malformed snapshot token: base64 decode failed")

    # --- Split into payload and signature ---
    # The payload contains colons only in integer and run_id fields; no dots.
    # Splitting on "." gives exactly ["payload", "signature"].
    parts = raw.split(".")
    if len(parts) != 2:
        raise ValueError("Malformed snapshot token: expected payload.signature")

    payload_str, received_sig = parts[0], parts[1]

    # --- Verify HMAC before parsing payload (fail-fast on tamper) ---
    secret = _get_secret()
    expected_sig = _sign_payload(payload_str, secret)
    if not hmac.compare_digest(expected_sig, received_sig):
        raise PermissionError("Invalid snapshot token signature")

    # --- Parse fields from verified payload ---
    # Accept 3-field (legacy) and 4-field (with run_id) payloads.
    fields = payload_str.split(":")
    if len(fields) not in (3, 4):
        raise ValueError(
            "Malformed snapshot token: payload must have three or four fields"
        )

    try:
        game_id        = int(fields[0])
        at_bat_index   = int(fields[1])
        revealed_count = int(fields[2])
    except ValueError:
        raise ValueError("Malformed snapshot token: payload fields must be integers")

    if game_id <= 0 or at_bat_index < 0 or revealed_count < 0:
        raise ValueError("Malformed snapshot token: field values out of range")

    # --- Extract and validate run_id if present ---
    run_id: str | None = None
    if len(fields) == 4:
        run_id = fields[3]
        if not run_id:
            raise ValueError("Malformed snapshot token: run_id field must not be empty")
        if not _RUN_ID_RE.match(run_id):
            raise ValueError(
                f"Malformed snapshot token: run_id must be lowercase hex, "
                f"got: {run_id!r}"
            )

    return game_id, at_bat_index, revealed_count, run_id

# ---------------------------------------------------------------------------
# POST /snapshot/redeem is the ONLY endpoint registered as a Pitch Scout tool
# in IBM Orchestrate.  It accepts a signed token and returns a spoiler-safe
# pitch prefix.  The existing GET /games/{game_id}/at-bats/{at_bat_index}
# endpoint is not registered as an agent tool.
#
# Token creation is strictly internal (snapshot.create_snapshot_token).
# No HTTP endpoint issues tokens.
#
# Authorization: revealed_count=0 is a valid token value (used internally for
# pregame fact sheets) but the IBM tool contract requires revealed_count >= 1.
# The Analyst prompt instructs Pitch Scout to only redeem tokens for valid
# pitch positions; the backend does not enforce a minimum of 1 here because
# the HMAC signature already guarantees the token was created by the backend.
# ---------------------------------------------------------------------------

class RedeemSnapshotRequest(BaseModel):
    token: str


@app.post("/snapshot/redeem")
def redeem_snapshot(
    request: RedeemSnapshotRequest,
    user_agent: str | None = Header(default=None),
):
    """
    Redeem a signed snapshot token and return a spoiler-safe pitch prefix.

    The token was created internally by create_snapshot_token and encodes
    (game_id, at_bat_index, revealed_count[, run_id]).  The backend verifies
    the HMAC before using any value from the token.

    Spoiler-safety guarantees
    -------------------------
    - Only pitches 1 … revealed_count are returned (enforced by mlb_data.py).
    - outcome is null unless revealed_count >= total_pitches (is_complete).
    - total_pitches is omitted when is_complete is false.
    - Entering-game AVG / ERA are included (pre-game, non-spoiler).
    - Full-game lines and postgame season totals are never included.

    Redemption history
    ------------------
    Each successful redemption is appended to the in-memory ring buffer
    (run_id included when present) and logged at INFO level.
    Tokens and secrets are never stored or logged.
    """
    # --- Verify token ---
    try:
        game_id, at_bat_index, revealed_count, run_id = verify_snapshot_token(
            request.token
        )
    except PermissionError:
        raise HTTPException(status_code=403, detail="Invalid snapshot token")
    except ValueError as exc:
        msg = str(exc)
        if "SNAPSHOT_SECRET" in msg:
            raise HTTPException(
                status_code=503, detail="Snapshot signing not configured"
            )
        raise HTTPException(
            status_code=400, detail=f"Malformed snapshot token: {msg}"
        )

    # --- Fetch MLB game data ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=mlb_error",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=502, detail="Unable to retrieve MLB game data"
        )

    # --- Build spoiler-safe snapshot ---
    try:
        snapshot = build_authorized_snapshot(
            game_data,
            game_id=game_id,
            at_bat_index=at_bat_index,
            revealed_count=revealed_count,
        )
    except DataError as exc:
        msg = str(exc)
        if "not found" in msg:
            logger.warning(
                "snapshot_redeem game=%d at_bat=%d count=%d status=not_found",
                game_id, at_bat_index, revealed_count,
            )
            raise HTTPException(status_code=404, detail="At-bat not found")
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=count_exceeds_total",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=404,
            detail="revealed_count exceeds available pitches",
        )

    # --- Record redemption event (no token, no secret) ---
    _record_redemption(game_id, at_bat_index, revealed_count, run_id, user_agent)

    logger.info(
        "snapshot_redeem ts=%s game=%d at_bat=%d authorized_count=%d "
        "run_id=%s is_complete=%s status=ok",
        datetime.now(timezone.utc).isoformat(),
        game_id,
        at_bat_index,
        revealed_count,
        run_id,  # safe to log: not a secret
        snapshot["is_complete"],
    )

    return snapshot

    This endpoint is NOT registered as an IBM agent tool.
    It returns all pitches and the final outcome (no spoiler-gating).

    Response keys (unchanged from previous implementation):
        game_id, at_bat_index, batter, pitcher, inning, half_inning,
        outcome, pitches, batter_stats, pitcher_stats

    MLB data is fetched exactly once per request.
    Stats extraction uses mlb_data.extract_batter_stats /
    extract_pitcher_stats (same logic as before, now shared).
    """
    if game_id <= 0 or at_bat_index < 0:
        raise HTTPException(status_code=400, detail="Invalid ID")

    # --- Single MLB fetch ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        raise HTTPException(
            status_code=502,
            detail="Unable to retrieve MLB game data",
        )

    # --- Locate the at-bat and collect pitch events directly ---
    # We do NOT use build_authorized_snapshot here because that function
    # enforces spoiler-safety (withholds outcome, omits total_pitches).
    # This public GET endpoint always returns the complete at-bat.
    plays = game_data.get("liveData", {}).get("plays", {}).get("allPlays", [])
    at_bat = next(
        (play for play in plays
         if play.get("about", {}).get("atBatIndex") == at_bat_index),
        None,
    )
    if at_bat is None:
        raise HTTPException(status_code=404, detail="At-bat not found")

    pitch_events = [
        event for event in at_bat.get("playEvents", [])
        if event.get("isPitch")
    ]

    pitches = []
    for event in pitch_events:
        details     = event.get("details") or {}
        pitch_data  = event.get("pitchData") or {}
        coordinates = pitch_data.get("coordinates") or {}
        pitches.append({
            "number":            event.get("pitchNumber"),
            "type":              (details.get("type") or {}).get("description"),
            "result":            details.get("description"),
            "velocity_mph":      pitch_data.get("startSpeed"),
            "plate_x":           coordinates.get("pX"),
            "plate_z":           coordinates.get("pZ"),
            "count_after_pitch": event.get("count"),
        })

    matchup = at_bat.get("matchup", {})
    about   = at_bat.get("about", {})

    # --- Entering-game stats (verified at-bats only) ---
    batter_stats  = None
    pitcher_stats = None
    if (game_id, at_bat_index) in VERIFIED_STATS_AT_BATS:
        batter_id  = matchup.get("batter", {}).get("id")
        pitcher_id = matchup.get("pitcher", {}).get("id")
        boxscore_teams = (
            game_data.get("liveData", {})
                     .get("boxscore", {})
                     .get("teams", {})
        )
        try:
            batter_stats = extract_batter_stats(boxscore_teams, batter_id)
        except Exception:
            batter_stats = None
        try:
            pitcher_stats = extract_pitcher_stats(boxscore_teams, pitcher_id)
        except Exception:
            pitcher_stats = None

    # Response keys are identical to the previous implementation.
    return {
        "game_id":       game_id,
        "at_bat_index":  at_bat_index,
        "batter":        matchup.get("batter", {}).get("fullName"),
        "pitcher":       matchup.get("pitcher", {}).get("fullName"),
        "inning":        about.get("inning"),
        "half_inning":   about.get("halfInning"),
        "outcome":       at_bat.get("result", {}).get("event"),
        "pitches":       pitches,
        "batter_stats":  batter_stats,
        "pitcher_stats": pitcher_stats,
    }

def redeem_snapshot(
    request: RedeemSnapshotRequest,
    user_agent: str | None = Header(default=None),
):
    """
    Redeem a signed snapshot token and return a spoiler-safe pitch prefix.

    The token was created internally by create_snapshot_token and encodes
    (game_id, at_bat_index, revealed_count[, run_id]).  The backend verifies
    the HMAC before using any value from the token.

    Spoiler-safety guarantees (enforced by mlb_data.build_authorized_snapshot)

    - outcome is null unless revealed_count >= total_pitches (is_complete).
    - total_pitches is omitted when is_complete is false.
    - Entering-game AVG / ERA are included (pre-game, non-spoiler).
    - Full-game lines and postgame season totals are never included.

    Redemption history
    ------------------
    Each successful redemption is appended to the in-memory ring buffer
    with run_id (when present) for correlation.  Tokens and secrets are
    never stored or logged.
    """
    # --- Verify token (HMAC checked before any payload values are used) ---
    try:
        game_id, at_bat_index, revealed_count, run_id = verify_snapshot_token(
            request.token
        )
    except PermissionError:
        raise HTTPException(status_code=403, detail="Invalid snapshot token")
    except ValueError as exc:
        if "SNAPSHOT_SECRET" in str(exc):
            raise HTTPException(
                status_code=503, detail="Snapshot signing not configured"
            )
        # Generic message: do not expose internal token structure to callers.
        raise HTTPException(status_code=400, detail="Malformed snapshot token")

    # --- Fetch MLB game data (single fetch) ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=mlb_error",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=502, detail="Unable to retrieve MLB game data"
        )

    # --- Build spoiler-safe snapshot (single authoritative call) ---
    try:
        snapshot = build_authorized_snapshot(
            game_data,
            game_id=game_id,
            at_bat_index=at_bat_index,
            revealed_count=revealed_count,
        )
    except DataError as exc:
        msg = str(exc)
        if "not found" in msg:
            logger.warning(
                "snapshot_redeem game=%d at_bat=%d count=%d status=not_found",
                game_id, at_bat_index, revealed_count,
            )
            raise HTTPException(status_code=404, detail="At-bat not found")
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=count_exceeds_total",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=404,
            detail="revealed_count exceeds available pitches",
        )

    # --- Record redemption event (run_id used for correlation; no token stored) ---
    _record_redemption(game_id, at_bat_index, revealed_count, run_id, user_agent)

    logger.info(
        "snapshot_redeem ts=%s game=%d at_bat=%d authorized_count=%d "
        "run_id=%s is_complete=%s status=ok",
        datetime.now(timezone.utc).isoformat(),
        game_id,
        at_bat_index,
        revealed_count,
        run_id,          # safe to log: not a secret
        snapshot["is_complete"],
    )

    return snapshot

@app.get("/games/{game_id}/at-bats/{at_bat_index}")
def get_at_bat(game_id: int, at_bat_index: int):
    """
    Return full at-bat data for the requested game and at-bat index.

    Not registered as an IBM agent tool.
    Always returns the complete at-bat (all pitches, final outcome).

    Response keys — exactly 10, identical to previous implementation:
        game_id, at_bat_index, batter, pitcher, inning, half_inning,
        outcome, pitches, batter_stats, pitcher_stats

    Single MLB network fetch per request.

    Zero-pitch at-bat note
    ----------------------
    build_authorized_snapshot() with revealed_count=0 returns outcome=None
    (Guard 3: spoiler-safe builder never reveals outcome when is_complete=False).
    For this full-GET endpoint we always want the recorded outcome regardless
    of pitch count, so we read it directly from game_data after the snapshot
    is built and substitute it.  This preserves the existing API contract
    without weakening the spoiler-safe builder.
    """
    if game_id <= 0 or at_bat_index < 0:
        raise HTTPException(status_code=400, detail="Invalid ID")

    # --- Single MLB fetch ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        raise HTTPException(
            status_code=502,
            detail="Unable to retrieve MLB game data",
        )

    # --- Determine actual pitch count (no spoiler gate needed here) ---
    total = count_pitches_in_at_bat(game_data, at_bat_index)
    if total is None:
        raise HTTPException(status_code=404, detail="At-bat not found")

    # --- Build full at-bat via shared builder (revealed_count == total) ---
    # For total > 0: is_complete=True, outcome included, correct pitch list.
    # For total == 0: is_complete=False (Guard 3), outcome=None — corrected below.
    try:
        snapshot = build_authorized_snapshot(
            game_data,
            game_id=game_id,
            at_bat_index=at_bat_index,
            revealed_count=total,
        )
    except DataError:
        # Should not occur: total came from count_pitches_in_at_bat on the
        # same game_data, so the at-bat exists and count is valid.
        raise HTTPException(status_code=404, detail="At-bat not found")

    # --- Resolve outcome for zero-pitch at-bats ---
    # For normal at-bats (total > 0), snapshot["outcome"] is already correct.
    # For zero-pitch at-bats, is_complete=False so the builder returns None.
    # Preserve the previous API contract by reading the raw outcome directly.
    if total == 0:
        plays = game_data.get("liveData", {}).get("plays", {}).get("allPlays", [])
        at_bat_raw = next(
            (p for p in plays
             if p.get("about", {}).get("atBatIndex") == at_bat_index),
            None,
        )
        outcome = at_bat_raw.get("result", {}).get("event") if at_bat_raw else None
    else:
        outcome = snapshot["outcome"]

    # --- Return exactly the original 10 public response keys ---
    # revealed_count, is_complete, and total_pitches are internal builder fields
    # and are intentionally excluded from this public endpoint.
    return {
        "game_id":       snapshot["game_id"],
        "at_bat_index":  snapshot["at_bat_index"],
        "batter":        snapshot["batter"],
        "pitcher":       snapshot["pitcher"],
        "inning":        snapshot["inning"],
        "half_inning":   snapshot["half_inning"],
        "outcome":       outcome,
        "pitches":       snapshot["pitches"],
        "batter_stats":  snapshot["batter_stats"],
        "pitcher_stats": snapshot["pitcher_stats"],
    }

class RedeemSnapshotRequest(BaseModel):
    token: str


@app.post("/snapshot/redeem")
def redeem_snapshot(
    request: RedeemSnapshotRequest,
    user_agent: str | None = Header(default=None),
):
    """
    Redeem a signed snapshot token and return a spoiler-safe pitch prefix.

    Spoiler-safety guarantees (enforced by mlb_data.build_authorized_snapshot)
    --------------------------------------------------------------------------
    - Only pitches 1 … revealed_count are returned.
    - outcome is null unless revealed_count >= total_pitches (is_complete).
    - total_pitches is omitted when is_complete is false.
    - Entering-game AVG / ERA are included (pre-game, non-spoiler).

    Redemption history
    ------------------
    Each successful redemption is appended to the in-memory ring buffer
    with run_id for correlation.  Tokens and secrets are never stored or logged.
    """
    # --- Verify token (HMAC checked before any payload values are used) ---
    try:
        game_id, at_bat_index, revealed_count, run_id = verify_snapshot_token(
            request.token
        )
    except PermissionError:
        raise HTTPException(status_code=403, detail="Invalid snapshot token")
    except ValueError as exc:
        if "SNAPSHOT_SECRET" in str(exc):
            raise HTTPException(
                status_code=503, detail="Snapshot signing not configured"
            )
        # Generic 400 — do not expose internal token structure details.
        raise HTTPException(status_code=400, detail="Malformed snapshot token")

    # --- Single MLB fetch ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=mlb_error",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=502, detail="Unable to retrieve MLB game data"
        )

    # --- Build spoiler-safe snapshot ---
    try:
        snapshot = build_authorized_snapshot(
            game_data,
            game_id=game_id,
            at_bat_index=at_bat_index,
            revealed_count=revealed_count,
        )
    except DataError as exc:
        msg = str(exc)
        if "not found" in msg:
            logger.warning(
                "snapshot_redeem game=%d at_bat=%d count=%d status=not_found",
                game_id, at_bat_index, revealed_count,
            )
            raise HTTPException(status_code=404, detail="At-bat not found")
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=count_exceeds_total",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=404,
            detail="revealed_count exceeds available pitches",
        )

    # --- Record redemption (run_id used for correlation; no token stored) ---
    _record_redemption(game_id, at_bat_index, revealed_count, run_id, user_agent)

    logger.info(
        "snapshot_redeem ts=%s game=%d at_bat=%d authorized_count=%d "
        "run_id=%s is_complete=%s status=ok",
        datetime.now(timezone.utc).isoformat(),
        game_id,
        at_bat_index,
        revealed_count,
        run_id,
        snapshot["is_complete"],
    )

    return snapshot

    Response keys — exactly 10, identical to previous implementation:

    try:
        game_data = fetch_game_data(game_id)

        raise HTTPException(
            status_code=502,
            detail="Unable to retrieve MLB game data",
        )

    # --- Determine actual pitch count ---
    total = count_pitches_in_at_bat(game_data, at_bat_index)
    if total is None:
        raise HTTPException(status_code=404, detail="At-bat not found")

    # --- Build full at-bat via shared builder (revealed_count == total) ---
    # For total > 0: is_complete=True, outcome included, correct pitch list.
    # For total == 0: is_complete=False (Guard 3), outcome=None — corrected below.

            game_id=game_id,
            at_bat_index=at_bat_index,

        raise HTTPException(status_code=404, detail="At-bat not found")

    # For normal at-bats (total > 0), snapshot["outcome"] is correct.

def redeem_snapshot(

    user_agent: str | None = Header(default=None),
):

    Redeem a signed snapshot token and return a spoiler-safe pitch prefix.

    Spoiler-safety guarantees (enforced by mlb_data.build_authorized_snapshot)
    --------------------------------------------------------------------------
    - Only pitches 1 … revealed_count are returned.
    - outcome is null unless revealed_count >= total_pitches (is_complete).
    - total_pitches is omitted when is_complete is false.
    - Entering-game AVG / ERA are included (pre-game, non-spoiler).

    Redemption history
    ------------------
    Each successful redemption is appended to the in-memory ring buffer
    with run_id for correlation.  Tokens and secrets are never stored or logged.
    """

    try:
        game_id, at_bat_index, revealed_count, run_id = verify_snapshot_token(
            request.token
        )
    except PermissionError:
        raise HTTPException(status_code=403, detail="Invalid snapshot token")
    except ValueError as exc:
        if "SNAPSHOT_SECRET" in str(exc):
            raise HTTPException(
                status_code=503, detail="Snapshot signing not configured"
            )
        # Generic 400 — do not expose internal token structure details.
        raise HTTPException(status_code=400, detail="Malformed snapshot token")

    # --- Single MLB fetch ---
    try:
        game_data = fetch_game_data(game_id)
    except NetworkError:
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=mlb_error",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=502, detail="Unable to retrieve MLB game data"
        )

    # --- Build spoiler-safe snapshot ---
    try:
        snapshot = build_authorized_snapshot(
            game_data,
            game_id=game_id,
            at_bat_index=at_bat_index,
            revealed_count=revealed_count,
        )
    except DataError as exc:
        msg = str(exc)
        if "not found" in msg:
            logger.warning(
                "snapshot_redeem game=%d at_bat=%d count=%d status=not_found",
                game_id, at_bat_index, revealed_count,
            )
            raise HTTPException(status_code=404, detail="At-bat not found")
        logger.warning(
            "snapshot_redeem game=%d at_bat=%d count=%d status=count_exceeds_total",
            game_id, at_bat_index, revealed_count,
        )
        raise HTTPException(
            status_code=404,
            detail="revealed_count exceeds available pitches",
        )

    # --- Record redemption (run_id used for correlation; no token stored) ---
    _record_redemption(game_id, at_bat_index, revealed_count, run_id, user_agent)

    logger.info(
        "snapshot_redeem ts=%s game=%d at_bat=%d authorized_count=%d "

        datetime.now(timezone.utc).isoformat(),
        game_id,
        at_bat_index,
        revealed_count,
        run_id,
        snapshot["is_complete"],
    )

    return snapshot

def _check_future_pitch_refs(text: str, revealed_count: int) -> list[int]:
    """
    Scan text for explicit future-pitch references.

    Returns a list of pitch numbers > revealed_count found in the text.
    Both ends of a range ("pitches N-M") are checked.
    Returns an empty list when no future references are found.
    """
    spoiler_refs = []
    for m in _FUTURE_PITCH_RE.finditer(text):
        n1 = int(m.group(1))
        if n1 > revealed_count:
            spoiler_refs.append(n1)
        if m.group(2) is not None:
            n2 = int(m.group(2))
            if n2 > revealed_count:
                spoiler_refs.append(n2)
    return spoiler_refs

            "Analyst response missing 'evidence_pitches' list. "
            f"Got keys: {list(parsed.keys())}",
            raw_response=raw_content,
        )

    if len(evidence_pitches) == 0:
        raise StageAError(
            "'evidence_pitches' must not be empty; "
            "the Analyst must cite at least one pitch from the snapshot.",
            raw_response=raw_content,
        )

    if not all(isinstance(p, int) for p in evidence_pitches):
        raise StageAError(
            f"'evidence_pitches' must be a list of integers, "
            f"got: {evidence_pitches!r}",
            raw_response=raw_content,
        )

    out_of_range = [p for p in evidence_pitches if p < 1 or p > revealed_count]
    if out_of_range:
        raise StageAError(
            f"'evidence_pitches' contains out-of-range values "
            f"(must be 1-{revealed_count}): {out_of_range}",
            raw_response=raw_content,
        )

    # --- Spoiler check: future-pitch references (single and range) ---
    # Uses _check_future_pitch_refs() which handles both "pitch N" and
    # "pitches N-M" / "pitches N–M" (en-dash) patterns.
    spoiler_refs = _check_future_pitch_refs(analysis, revealed_count)
    if spoiler_refs:
        raise StageAError(
            f"Analyst analysis references future pitch number(s) "
            f"beyond revealed_count={revealed_count}: {spoiler_refs}",
            raw_response=raw_content,
        )

    return {
        "analysis": analysis.strip(),
        "evidence_pitches": evidence_pitches,
    }

def build_fact_sheet(
    game_data: dict,
    game_id: int,
    at_bat_index: int,
    revealed_count: int,
) -> str:
    """
    Build a plain-text verified fact sheet from already-fetched MLB game data.

    All values come from runtime MLB data via build_authorized_snapshot().
    No velocities, statistics, counts, or outcomes are hardcoded.

    Respects the same spoiler gate as /snapshot/redeem:
    - Only pitches 1 … revealed_count are included.
    - Final outcome is omitted unless is_complete is True.
    - Entering-game stats (AVG, ERA) are included when available;
      they are pre-game values and do not spoil the replay.

    Format: "Pitches revealed so far: N" with no "of M" phrasing.

    SOURCE NOTE — velocities (e.g. 91.9, 93.1, 93.6 mph):
    The values previously cited in IBM-generated Stage A analysis (stage_b_test.py
    fixture) came from the IBM Analyst's output, not from independent verification
    of the raw MLB data.  This function reads velocity_mph directly from
    build_authorized_snapshot() → MLB Stats API → pitchData.startSpeed.
    Those runtime values supersede any previously quoted numbers.

    Parameters
    ----------
    game_data      : full MLB Stats API response from fetch_game_data()
    game_id        : MLB game identifier
    at_bat_index   : zero-based at-bat index
    revealed_count : number of pitches authorized (>= 0)

    Returns
    -------
    Plain-text fact sheet string.

    Raises
    ------
    DataError : if revealed_count is invalid or at-bat not found.
    """
    snapshot = build_authorized_snapshot(
        game_data,
        game_id=game_id,
        at_bat_index=at_bat_index,
        revealed_count=revealed_count,
    )

    lines = []

    batter  = snapshot["batter"]  or "Unknown batter"
    pitcher = snapshot["pitcher"] or "Unknown pitcher"
    lines.append(f"Batter: {batter}")
    lines.append(f"Pitcher: {pitcher}")

    bs = snapshot.get("batter_stats")
    if bs:
        lines.append(
            f"Batter entering-game AVG: {bs['avg']:.3f} "
            f"({bs['hits']} H / {bs['at_bats']} AB)"
        )
    ps = snapshot.get("pitcher_stats")
    if ps:
        lines.append(
            f"Pitcher entering-game ERA: {ps['era']:.2f} "
            f"({ps['earned_runs']} ER / {ps['outs_recorded']} outs)"
        )

    lines.append("")
    lines.append(f"Pitches revealed so far: {revealed_count}")

    pitches = snapshot["pitches"]
    if pitches:
        for p in pitches:
            count = p.get("count_after_pitch") or {}
            count_str = f"{count.get('balls', '?')}-{count.get('strikes', '?')}"
            vel = p.get("velocity_mph")
            vel_str = f"{vel} mph" if vel is not None else "velocity unavailable"
            lines.append(
                f"  Pitch {p['number']}: {p['type'] or 'Unknown type'}, "
                f"{vel_str}, result: {p['result'] or 'Unknown result'}, "
                f"count after: {count_str}"
            )
    else:
        lines.append("  (no pitches revealed yet)")

    lines.append("")

    if pitches:
        last_count = pitches[-1].get("count_after_pitch") or {}
        lines.append(
            f"Current count: "
            f"{last_count.get('balls', '?')}-{last_count.get('strikes', '?')}"
        )
    else:
        lines.append("Current count: 0-0 (no pitches yet)")

    if snapshot["is_complete"] and snapshot["outcome"]:
        lines.append(f"Final at-bat outcome: {snapshot['outcome']}")
    else:
        lines.append("At-bat outcome: not yet revealed")

    return "\n".join(lines)

_FAN_EXPERIENCE_PROMPT_TEMPLATE = """\
You are DiamondIQ Fan Experience. Write four baseball broadcast commentaries \
\u2014 one per knowledge level \u2014 based strictly on the ANALYSIS and VERIFIED \
FACT SHEET below. Both sources are authoritative; facts are identical across \

STRICT ACCURACY RULES (apply to every level):
- Use only facts present in the ANALYSIS and FACT SHEET. Do not invent any \
pitches, results, velocities, counts, or outcomes not listed there.
- Only mention the final outcome if it is explicitly stated.
- Do not describe a strike as "called" or "swinging" unless that \
classification is explicitly stated.
- Do not describe pitch location (e.g. "low", "inside", "high", \
"flirted with the zone") unless the ANALYSIS explicitly uses that description.
- Do not claim pitcher intent (e.g. "trying to", "setting up", "attempting").
- Never imply that a ball resets, neutralises, or erases a strike count. A \
ball advances the ball count by one; the strikes already recorded remain. \
Forbidden phrases: "fresh chance", "reset the count", "back in the count", \
"leveled the count" when describing a ball after two strikes.
- Do not say the pitcher is "ahead" when the count favours the batter, or \

- Do not reveal future pitches or outcomes.

KNOWLEDGE LEVEL REQUIREMENTS:

first time.
- learning (45-60 words): Assumes reader knows balls, strikes, and outs. Name \
pitch types briefly. Explain how the current count affects the situation. Add \
one sentence of significance or context.
- regular (50-65 words): Natural broadcast voice. Use pitch-type names freely. \
Describe count progression and any momentum shift. No explanations of basic \
terms. Build anticipation without revealing future events.
- expert (55-70 words): Use verified velocities and pitch types from the FACT \
SHEET when present. Reference entering-game AVG or ERA when in the FACT SHEET. \
Analyze the observable count context and pitch-type pattern. Distinguish what \
is observed from what is inferred. Never invent strategic decisions.

ANALYSIS:
{stage_a_analysis}

VERIFIED FACT SHEET:
{fact_sheet}

Respond with valid JSON only \u2014 no prose before or after, no Markdown fences. \
Use exactly this structure:

{{
  "commentary_by_level": {{
    "novice": "...",
    "learning": "...",
    "regular": "...",
    "expert": "..."
  }}
}}

All four commentaries must be grounded exclusively in the ANALYSIS and \
VERIFIED FACT SHEET above.
"""


class FanCommentaryError(ValueError):
    """Raised when Fan Experience returns an invalid or incomplete response.

    Attributes:
        raw_response: the unmodified string returned by the IBM agent.
                      Never include this in log output or HTTP responses.
    """
    def __init__(self, message: str, raw_response: str = "") -> None:
        super().__init__(message)
        self.raw_response = raw_response


def _strip_fences(text: str) -> str:
    """Remove surrounding whitespace and optional Markdown code fences."""
    text = text.strip()
    m = _re.match(r"^```(?:json)?\s*([\s\S]*?)\s*```$", text)
    if m:
        return m.group(1).strip()
    return text


def _validate_stage_b_output(
    commentary: dict,
    revealed_count: int,
) -> list[str]:
    """
    Post-generation safeguard scan on Stage B commentary text.

    Returns a list of violation strings (empty = clean).  Does NOT guarantee
    factual accuracy — it catches only explicit structural violations:
      1. Future pitch number references (e.g. "pitch 5" when revealed=3).
      2. Explicitly forbidden reset-count phrases.

    This is a defence-in-depth measure; prompt instructions are the primary
    control.  Regex cannot detect all factual errors.
    """
    _FORBIDDEN_PHRASES = (
        "fresh chance",
        "reset the count",
        "back in the count",
        "leveled the count",
    )
    violations = []
    for level, text in commentary.items():
        # Future-pitch references
        future = _check_future_pitch_refs(text, revealed_count)
        if future:
            violations.append(
                f"{level}: future pitch reference(s) {future}"
            )
        # Explicitly forbidden phrases
        text_lower = text.lower()
        for phrase in _FORBIDDEN_PHRASES:
            if phrase in text_lower:
                violations.append(
                    f"{level}: forbidden phrase {phrase!r}"
                )
    return violations


def generate_fan_commentary(
    stage_a_input: dict,
    fact_sheet: str = "",
    revealed_count: int = 0,
) -> dict:
    """
    Call Fan Experience directly (no Analyst, no Pitch Scout) with verified
    Stage A data and a verified fact sheet.

    Args:
        stage_a_input: dict with keys:
            - "analysis" (str): verified Stage A analysis text
            - "evidence_pitches" (list[int]): pitch numbers from the
              verified snapshot; attached to the result unchanged.
        fact_sheet: plain-text verified fact sheet from build_fact_sheet().
                    May be an empty string; the prompt will include an empty
                    VERIFIED FACT SHEET section in that case.
        revealed_count: number of pitches in the authorized window; used for
                    post-generation spoiler validation.  Defaults to 0
                    (skips range check) for backward compatibility.

    Returns:
        dict with keys:
            - "commentary_by_level": {"novice": str, "learning": str,
                                      "regular": str, "expert": str}
            - "evidence_pitches": list[int] (passed through from input)

    Raises:
        TypeError: if stage_a_input is not a dict or required keys are
                   the wrong type.
        KeyError: if WO_API_KEY, WO_INSTANCE_URL, or
                  WO_FAN_EXPERIENCE_AGENT_ID are not set.
        FanCommentaryError: if the IBM response is missing, malformed,
                            incomplete, or fails post-generation validation.
    """
    if not isinstance(stage_a_input, dict):
        raise TypeError(
            f"stage_a_input must be a dict, got {type(stage_a_input).__name__}"
        )
    analysis = stage_a_input.get("analysis")
    evidence_pitches = stage_a_input.get("evidence_pitches")
    if not isinstance(analysis, str) or not analysis.strip():
        raise TypeError(
            "stage_a_input['analysis'] must be a non-empty string"
        )
    if not isinstance(evidence_pitches, list) or not all(
        isinstance(p, int) for p in evidence_pitches
    ):
        raise TypeError(
            "stage_a_input['evidence_pitches'] must be a list of ints"
        )
    if not isinstance(fact_sheet, str):
        raise TypeError(
            f"fact_sheet must be a string, got {type(fact_sheet).__name__}"
        )

    api_key = os.environ["WO_API_KEY"]
    instance_url = os.environ["WO_INSTANCE_URL"].rstrip("/")
    fan_agent_id = os.environ["WO_FAN_EXPERIENCE_AGENT_ID"]

    token_data = urlencode({
        "grant_type": "urn:ibm:params:oauth:grant-type:apikey",
        "apikey": api_key
    }).encode("utf-8")

    token_request = Request(
        "https://iam.cloud.ibm.com/identity/token",
        data=token_data,
        headers={"Content-Type": "application/x-www-form-urlencoded"},
        method="POST"
    )

    with urlopen(token_request, timeout=20) as response:
        access_token = json.load(response)["access_token"]

    prompt = _FAN_EXPERIENCE_PROMPT_TEMPLATE.format(
        stage_a_analysis=analysis,
        fact_sheet=fact_sheet,
    )

    payload = json.dumps({
        "messages": [{"role": "user", "content": prompt}],
        "stream": False
    }).encode("utf-8")

    agent_request = Request(
        f"{instance_url}/v1/orchestrate/{fan_agent_id}/chat/completions",
        data=payload,
        headers={
            "Authorization": f"Bearer {access_token}",
            "Content-Type": "application/json"
        },
        method="POST"
    )

    with urlopen(agent_request, timeout=120) as response:
        result = json.load(response)

    raw_content = result["choices"][0]["message"]["content"]

    cleaned = _strip_fences(raw_content)

    try:
        parsed = json.loads(cleaned)
    except json.JSONDecodeError as exc:
        raise FanCommentaryError(
            f"Fan Experience returned non-JSON content: {exc}",
            raw_response=raw_content,
        ) from exc

    if not isinstance(parsed, dict):
        raise FanCommentaryError(
            f"Fan Experience JSON root must be a dict, got "
            f"{type(parsed).__name__}",
            raw_response=raw_content,
        )

    commentary = parsed.get("commentary_by_level")
    if not isinstance(commentary, dict):
        raise FanCommentaryError(
            "Fan Experience response missing 'commentary_by_level' dict. "
            f"Got keys: {list(parsed.keys())}",
            raw_response=raw_content,
        )

    missing = [lvl for lvl in _FAN_EXPERIENCE_LEVELS if lvl not in commentary]
    if missing:
        raise FanCommentaryError(
            f"Fan Experience response missing required level(s): {missing}",
            raw_response=raw_content,
        )

    empty = [
        lvl for lvl in _FAN_EXPERIENCE_LEVELS
        if not isinstance(commentary[lvl], str) or not commentary[lvl].strip()
    ]
    if empty:
        raise FanCommentaryError(
            f"Fan Experience returned empty commentary for level(s): {empty}",
            raw_response=raw_content,
        )

    # --- Post-generation safeguard validation ---
    # Check only when revealed_count > 0 (caller opted in).
    # This catches explicit structural violations; it does not guarantee
    # full factual accuracy.
    if revealed_count > 0:
        violations = _validate_stage_b_output(commentary, revealed_count)
        if violations:
            raise FanCommentaryError(
                f"Fan Experience output failed post-generation validation: "
                f"{violations}",
                raw_response=raw_content,
            )

    parsed["evidence_pitches"] = evidence_pitches
    return parsed

## 4. Snapshot Authorization & Spoiler Prevention

### Token format
```
base64url( "game_id:at_bat_index:revealed_count[:{run_id}]" + "." + hex_hmac_sha256 )
```

**Legacy (3-field):** `game_id:at_bat_index:revealed_count`  
**New (4-field):** `game_id:at_bat_index:revealed_count:run_id` (run_id = 16 lowercase hex chars)

Both formats verify correctly. `verify_snapshot_token()` now returns a 4-tuple:
```python
game_id, at_bat_index, revealed_count, run_id = verify_snapshot_token(token)
# run_id is None for legacy tokens
```

### Spoiler gates (enforced in `mlb_data.build_authorized_snapshot()`)
1. `revealed_count` must be a non-negative integer (not bool, float, str)
2. Only pitches 1…revealed_count returned
3. `outcome` is `None` unless `is_complete` is `True`
4. `is_complete = (total_pitches > 0 and revealed_count >= total_pitches)` — zero-pitch at-bats never auto-complete
5. `total_pitches` omitted from response when `is_complete` is `False`
6. Out-of-range error messages do not reveal `total_pitches`
